cmake_minimum_required(VERSION 3.16)
project(lowtide C)
set(CMAKE_C_STANDARD 99)
option(LT_R36S "Build for R36S/R36H (640x480, 3x scale, gamepad-only, aarch64 Linux)" OFF)
option(LT_STATIC_SDL "Link SDL2 statically when possible" OFF)

# 1) content pipeline -> build/gen/blob.c (Python 3 + Pillow-free; numpy only for music previews)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GEN ${CMAKE_BINARY_DIR}/gen)
file(GLOB_RECURSE LT_DATA_DEPS ${CMAKE_SOURCE_DIR}/tools/*.py ${CMAKE_SOURCE_DIR}/content/*.py ${CMAKE_SOURCE_DIR}/core/layout.h)
add_custom_command(OUTPUT ${GEN}/blob.c ${GEN}/gen_strings.h
  COMMAND ${CMAKE_COMMAND} -E make_directory ${GEN}
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/build_data.py --out ${GEN}
  DEPENDS ${LT_DATA_DEPS} WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_custom_target(lt_data DEPENDS ${GEN}/blob.c)

# 2) game
file(GLOB CORE_SRC ${CMAKE_SOURCE_DIR}/core/*.c)
add_executable(lowtide ${CORE_SRC} platform/sdl/sdl_main.c platform/soft/soft.c ${GEN}/blob.c)
add_dependencies(lowtide lt_data)
target_include_directories(lowtide PRIVATE core ${GEN})
if(NOT SDL2_FOUND AND NOT TARGET SDL2::SDL2)
  find_package(SDL2 CONFIG QUIET)
endif()
if(TARGET SDL2::SDL2-static AND NOT TARGET SDL2::SDL2)
  target_link_libraries(lowtide PRIVATE SDL2::SDL2-static)
elseif(TARGET SDL2::SDL2)
  if(WIN32 AND TARGET SDL2::SDL2main)
    target_link_libraries(lowtide PRIVATE mingw32 SDL2::SDL2main SDL2::SDL2)
  else()
    target_link_libraries(lowtide PRIVATE SDL2::SDL2)
  endif()
elseif(SDL2_INCLUDE_DIRS)
  target_include_directories(lowtide PRIVATE ${SDL2_INCLUDE_DIRS})
  target_link_libraries(lowtide PRIVATE ${SDL2_LIBRARIES})
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED sdl2)
  target_include_directories(lowtide PRIVATE ${SDL2_INCLUDE_DIRS})
  target_link_libraries(lowtide PRIVATE ${SDL2_LIBRARIES})
endif()
if(NOT WIN32)
  target_link_libraries(lowtide PRIVATE m)
endif()
if(LT_R36S)
  target_compile_definitions(lowtide PRIVATE TARGET_R36S=1)
endif()
if(WIN32)
  target_link_options(lowtide PRIVATE -mwindows -static-libgcc)
endif()
target_compile_options(lowtide PRIVATE -O2 -Wall -Wno-unused-function -Wno-misleading-indentation)
install(TARGETS lowtide RUNTIME DESTINATION .)
